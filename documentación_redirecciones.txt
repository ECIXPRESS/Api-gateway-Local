# EJEMPLOS DE COMANDOS PARA API GATEWAY - ECI EXPRESS
# Puerto: 8081 | URL Base: http://localhost:8081

## =============================================
## 1. AUTENTICACION - AuthController
## =============================================

# Login de Usuario (obtener token)
$loginBody = @{
    email = "manuelalejandro.guarnizo@gmail.com"
    password = "Valentina1234"
    role = "CUSTOMER"
} | ConvertTo-Json

Write-Host "=== INICIANDO LOGIN ==="
$loginResponse = Invoke-RestMethod -Uri "http://localhost:8081/api/auth/login" -Method Post -ContentType "application/json" -Body $loginBody
Write-Host "Login exitoso!"
Write-Host "Token de acceso: $($loginResponse.accessToken)"


# Guardar el token para usar en otros requests
$accessToken = $loginResponse.accessToken
$headers = @{
    "Authorization" = "Bearer $accessToken"
    "Content-Type" = "application/json"
}

# Validar Token
Write-Host "=== VALIDANDO TOKEN ==="
$validationResponse = Invoke-RestMethod -Uri "http://localhost:8081/api/auth/validate?token=$accessToken" -Method Get
Write-Host "Token valido: $($validationResponse.valid)"

# Extraer Email del Token
Write-Host "=== EXTRAYENDO EMAIL DEL TOKEN ==="
$emailResponse = Invoke-RestMethod -Uri "http://localhost:8081/api/auth/extract-username?token=$accessToken" -Method Get
Write-Host "Email extraido: $($emailResponse.email)"

# Obtener Usuario Actual (requiere token)
Write-Host "=== OBTENIENDO USUARIO ACTUAL ==="
$userInfo = Invoke-RestMethod -Uri "http://localhost:8081/api/auth/me" -Method Get -Headers $headers
Write-Host "Usuario actual: $($userInfo.email)"
Write-Host "Rol: $($userInfo.role)"

## 2. ADMINISTRADORES - AdminController
## =============================================

# Crear Administrador (requiere permisos de admin)
Write-Host "=== CREANDO ADMINISTRADOR ==="
$adminBody = @{
    name = "Admin Principal"
    email = "admin@eci.com"
    password = "admin123"
    role = "ADMIN"
} | ConvertTo-Json

$newAdmin = Invoke-RestMethod -Uri "http://localhost:8081/api/users/admins" -Method Post -Headers $headers -Body $adminBody
Write-Host "Administrador creado: $($newAdmin.name)"
Write-Host "ID: $($newAdmin.id)"
Write-Host "Email: $($newAdmin.email)"

# Obtener Administrador por ID
Write-Host "=== OBTENIENDO ADMINISTRADOR ==="
$adminId = $newAdmin.id
$admin = Invoke-RestMethod -Uri "http://localhost:8081/api/users/admins/$adminId" -Method Get -Headers $headers
Write-Host "Administrador: $($admin.name)"
Write-Host "Email: $($admin.email)"

# Actualizar Administrador
Write-Host "=== ACTUALIZANDO ADMINISTRADOR ==="
$updateAdminBody = @{
    name = "Admin Actualizado"
    email = "admin.actualizado@eci.com"
} | ConvertTo-Json

$updatedAdmin = Invoke-RestMethod -Uri "http://localhost:8081/api/users/admins/$adminId" -Method Put -Headers $headers -Body $updateAdminBody
Write-Host "Administrador actualizado: $($updatedAdmin.name)"
Write-Host "Nuevo email: $($updatedAdmin.email)"

# Eliminar Administrador
Write-Host "=== ELIMINANDO ADMINISTRADOR ==="
Invoke-RestMethod -Uri "http://localhost:8081/api/users/admins/$adminId" -Method Delete -Headers $headers
Write-Host "Administrador eliminado"
